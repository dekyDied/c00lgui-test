-- // God Mode
local godConnection
local function ToggleGodMode(enabled)
    local player = Players.LocalPlayer
    if not player.Character or not player.Character:FindFirstChild("Humanoid") then return end
    local humanoid = player.Character.Humanoid
    
    if enabled then
        godConnection = humanoid.HealthChanged:Connect(function(health)
            if health < humanoid.MaxHealth then
                humanoid.Health = humanoid.MaxHealth
            end
        end)
        WindUI:Notify({ Title = "God Mode", Content = "Enabled - You're immortal!", Icon = "shield", Duration = 3 })
    else
        if godConnection then
            godConnection:Disconnect()
            godConnection = nil
        end
        WindUI:Notify({ Title = "God Mode", Content = "Disabled", Icon = "shield", Duration = 3 })
    end
end

-- // Infinite Ammo (Basic tool manipulation)
local ammoConnection
local function ToggleInfiniteAmmo(enabled)
    local player = Players.LocalPlayer
    if enabled then
        ammoConnection = RunService.Heartbeat:Connect(function()
            for _, tool in ipairs(player.Backpack:GetChildren()) do
                if tool:IsA("Tool") and tool:FindFirstChild("Ammo") then
                    tool.Ammo.Value = math.huge
                end
            end
            if player.Character and player.Character:FindFirstChildOfClass("Tool") then
                local equippedTool = player.Character:FindFirstChildOfClass("Tool")
                if equippedTool and equippedTool:FindFirstChild("Ammo") then
                    equippedTool.Ammo.Value = math.huge
                end
            end
        end)
        WindUI:Notify({ Title = "Infinite Ammo", Content = "Enabled - Never run out!", Icon = "gun", Duration = 3 })
    else
        if ammoConnection then
            ammoConnection:Disconnect()
            ammoConnection = nil
        end
        WindUI:Notify({ Title = "Infinite Ammo", Content = "Disabled", Icon = "gun", Duration = 3 })
    end
end

-- // Zombie ESP (Simple highlight for zombies)
local espConnections = {}
local function ToggleZombieESP(enabled)
    for _, conn in ipairs(espConnections) do
        conn:Disconnect()
    end
    espConnections = {}
    
    if enabled then
        for _, obj in ipairs(workspace:GetDescendants()) do
            if obj:IsA("Model") and (obj.Name:lower():find("zombie") or obj:FindFirstChild("Humanoid")) and obj ~= Players.LocalPlayer.Character then
                local highlight = Instance.new("Highlight")
                highlight.Parent = obj
                highlight.FillColor = Color3.fromRGB(255, 0, 0)
                highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
                
                local conn = obj.AncestryChanged:Connect(function()
                    if not obj.Parent then
                        highlight:Destroy()
                    end
                end)
                table.insert(espConnections, conn)
            end
        end
        
        -- Watch for new zombies
        local spawnConn = workspace.DescendantAdded:Connect(function(desc)
            if desc:IsA("Model") and (desc.Name:lower():find("zombie") or desc:FindFirstChild("Humanoid")) and desc ~= Players.LocalPlayer.Character then
                wait(0.1) -- Small delay
                local highlight = Instance.new("Highlight")
                highlight.Parent = desc
                highlight.FillColor = Color3.fromRGB(255, 0, 0)
                highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
            end
        end)
        table.insert(espConnections, spawnConn)
        
        WindUI:Notify({ Title = "Zombie ESP", Content = "Enabled - See all zombies!", Icon = "eye", Duration = 3 })
    else
        for _, obj in ipairs(workspace:GetDescendants()) do
            if obj:FindFirstChild("Highlight") and (obj.Parent.Name:lower():find("zombie") or obj.Parent:FindFirstChild("Humanoid")) then
                obj.Highlight:Destroy()
            end
        end
        WindUI:Notify({ Title = "Zombie ESP", Content = "Disabled", Icon = "eye", Duration = 3 })
    end
end

-- // Auto Fire (Simple mouse click simulation)
local autoFireConnection
local function ToggleAutoFire(enabled)
    if enabled then
        autoFireConnection = RunService.Heartbeat:Connect(function()
            if UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) then
                -- Simulate fire (assumes standard tool)
                local player = Players.LocalPlayer
                if player.Character and player.Character:FindFirstChildOfClass("Tool") then
                    mouse1click() -- Basic click simulation
                end
            end
        end)
        WindUI:Notify({ Title = "Auto Fire", Content = "Enabled - Hold to spray!", Icon = "crosshair", Duration = 3 })
    else
        if autoFireConnection then
            autoFireConnection:Disconnect()
            autoFireConnection = nil
        end
        WindUI:Notify({ Title = "Auto Fire", Content = "Disabled", Icon = "crosshair", Duration = 3 })
    end
end

-- // Config Tab
local configSection = Tabs.Config:Section({ Title = "Configuration Manager" })

local configFolder = "ZomBFiedHub/Configs"
makefolder(configFolder)

local function SaveConfig(fileName, data)
    local filePath = configFolder .. "/" .. fileName .. ".json"
    local jsonData = HttpService:JSONEncode(data)
    writefile(filePath, jsonData)
    WindUI:Notify({
        Title = "Saved",
        Content = "Configuration saved: " .. fileName,
        Icon = "save",
        Duration = 3
    })
end

local function LoadConfig(fileName)
    local filePath = configFolder .. "/" .. fileName .. ".json"
    if isfile(filePath) then
        local jsonData = readfile(filePath)
        return HttpService:JSONDecode(jsonData)
    end
    return nil
end

local function ListConfigs()
    local files = {}
    for _, file in ipairs(listfiles(configFolder)) do
        local name = file:match("^.+/(.+)%.json$")
        if name then
            table.insert(files, name)
        end
    end
    return files
end

local configName = ""
configSection:Input({
    Title = "Config Name",
    Placeholder = "Enter config name",
    Callback = function(text)
        configName = text
    end
})

configSection:Button({
    Title = "Save Config",
    Icon = "save",
    Callback = function()
        if configName ~= "" then
            SaveConfig(configName, {
                theme = WindUI:GetCurrentTheme(),
                transparency = WindUI:GetTransparency(),
                windowSize = { x = WindUI:GetWindowSize().X.Offset, y = WindUI:GetWindowSize().Y.Offset },
                cheats = Cheats,
                lastSave = os.date("%Y-%m-%d %H:%M:%S")
            })
        else
            WindUI:Notify({
                Title = "Error",
                Content = "Please enter a config name",
                Icon = "x",
                Duration = 3
            })
        end
    end
})

local configDropdown
configDropdown = configSection:Dropdown({
    Title = "Select Config",
    Multi = false,
    AllowNone = true,
    Values = ListConfigs(),
    Callback = function(selected)
        configName = selected
    end
})

configSection:Button({
    Title = "Load Config",
    Icon = "refresh-cw",
    Callback = function()
        if configName ~= "" then
            local data = LoadConfig(configName)
            if data then
                WindUI:SetTheme(data.theme)
                Window:ToggleTransparency(data.transparency)
                Window:Resize(UDim2.fromOffset(data.windowSize.x, data.windowSize.y))
                if data.cheats then
                    for cheat, enabled in pairs(data.cheats) do
                        Cheats[cheat] = enabled
                        if cheat == "GodMode" then ToggleGodMode(enabled) end
                        if cheat == "InfiniteAmmo" then ToggleInfiniteAmmo(enabled) end
                        if cheat == "SpeedBoost" then ToggleSpeedBoost(enabled) end
                        if cheat == "InfiniteJump" then ToggleInfiniteJump(enabled) end
                        if cheat == "ZombieESP" then ToggleZombieESP(enabled) end
                        if cheat == "AutoFire" then ToggleAutoFire(enabled) end
                    end
                end
                WindUI:Notify({
                    Title = "Loaded",
                    Content = "Configuration loaded: " .. configName,
                    Icon = "check",
                    Duration = 3
                })
                configDropdown:UpdateValues(ListConfigs())
            else
                WindUI:Notify({
                    Title = "Error",
                    Content = "Failed to load config",
                    Icon = "x",
                    Duration = 3
                })
            end
        else
            WindUI:Notify({
                Title = "Error",
                Content = "Please select a config",
                Icon = "x",
                Duration = 3
            })
        end
    end
})
